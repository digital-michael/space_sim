/*

Demo the selected solar system.

For-loop slice syntax (optional, append to group name):
  for $x in planets[3:]    skip first 3, visit the rest
  for $x in planets[:10]   first 10 items (indices 0–9)
  for $x in planets[3:10]  items at indices 3–9
  for $x in planets[-5:]   last 5 items

*/

hud off
hud info on
label nearest
sync on

//window size 640x480
window full on

set $orbits 1 // orbits around target
set $speed 20.0 // how fast to travel
set $sleep 1.5 // duration per item

sleep $sleep

for $x in stars
   nav jump "$x" $speed
   orbit "$x" $speed $orbits
   sleep $sleep
endfor

for $x in planets
    nav jump "$x" $speed
    //sleep $sleep
    orbit "$x" $speed $orbits
    sleep $sleep
endfor

for $x in dwarf_planets
    nav jump "$x" $speed
    //sleep $sleep
    orbit "$x" $speed $orbits
    sleep $sleep
endfor

for $x in moons[:10]
    nav jump "$x" $speed
    orbit "$x" $speed $orbits
    sleep $sleep
endfor

nav jump Sol $speed

sleep 5

//window restore
//window size 640x480
sync off
shutdown
exit
